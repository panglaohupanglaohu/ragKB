  Drives a multi-turn conversation with DeepSeek V4 where each turn the model can
  call tools (read_file, grep, write_file, patch_file, run_python, run_pytest) to

          path = (parsed.path or "").rstrip("/") + "/chat/completions"

              "messages": self.messages,
              "tools": self.tools,
              "tool_choice": "auto",
              "max_tokens": self.max_tokens,
              "temperature": self.temperature,
              "stream": False,
          })
          headers = {
              "Authorization": f"Bearer {self.api_key}",
              "Content-Type": "application/json",
          }
  
          last_err: Optional[Exception] = None
          for attempt in range(self._API_MAX_RETRIES):
              try:
                  conn = conn_cls(host, port, context=ctx, timeout=300) if ctx \
                      else conn_cls(host, port, timeout=300)
                  conn.request("POST", path, body=body_str, headers=headers)
                  resp = conn.getresponse()
                  raw = resp.read().decode("utf-8", errors="replace")
                  conn.close()
                  if resp.status == 429 or resp.status >= 500:
                      # Server-side error — retryable
                      raise RuntimeError(f"LLM HTTP {resp.status}: {raw[:300]}")
                  if resp.status >= 400:
                      raise RuntimeError(f"LLM HTTP {resp.status}: {raw[:500]}")
                  return json.loads(raw)

                  last_err = e

                  logger.warning(
                      "[AgentLoop] Server error on attempt %d/%d: %s — retrying in %ds",
                      attempt + 1, self._API_MAX_RETRIES, e, wait,
                  )
                  time.sleep(wait)
          raise last_err or RuntimeError("_post_chat failed after retries")

          self.messages.append({"role": "user", "content": user_prompt})
          self._emit("loop_start", {"role": self.role, "tools": [t["function"]["name"] for t in self.tools]})
  
          for it in range(self.max_iterations):
              # ── Safeguard 1: nudge agent when approaching iteration cap ──
              self._maybe_inject_nudge(it)

              self._compact_old_tool_results()

                  self._emit("error", {"iteration": it, "error": str(e)})
                  # If we have already done useful work, don't discard it —

                  if self.files_changed or self.summary:
                      logger.info(

                          len(self.files_changed), len(self.summary),

                          "files_changed": self.files_changed,
                          "iterations": it, "log": self.tool_call_log,

                  return {

                      "summary": self.summary, "files_changed": self.files_changed,

                  }

              tool_calls = msg.get("tool_calls") or []
              finish_reason = choice.get("finish_reason", "")
  
              self._emit("model_turn", {
                  "iteration": it,
                  "elapsed": round(time.time() - t0, 2),
                  "content_chars": len(content),
                  "tool_call_count": len(tool_calls),

任务 `manual-start-check` 和 `validate pending start path` 的核心需求是：

- **验证待启动路径**：检查系统中所有处于 "pending"（待启动/待确认）状态的项目或路径，验证其是否可以正常启动

@app.post("/api/v1/manual-start-check")
async def manual_start_check():
    """手动触发系统启动检查，验证所有子系统是否就绪。"""
    from channels.marine_base import get_default_registry
    registry = get_default_registry()
    
    results = {
        "timestamp": datetime.utcnow().isoformat(),
        "checks": {}
    }
    
    # 1. 检查演进引擎
    evo = registry.get("system_evolution")
    if evo:
        try:
            status = evo.get_status()
            results["checks"]["evolution_engine"] = {
                "status": "ready" if status.get("initialized") else "pending",
                "details": status
            }
        except Exception as e:
            results["checks"]["evolution_engine"] = {
                "status": "error",
                "error": str(e)
            }
    else:
        results["checks"]["evolution_engine"] = {
            "status": "not_registered"
        }
    
    # 2. 检查聊天通道
    chat = registry.get("bridge_chat")
    if chat:
        try:
            status = chat.get_status() if hasattr(chat, 'get_status') else {"initialized": True}
            results["checks"]["bridge_chat"] = {
                "status": "ready" if status.get("initialized") else "pending",
                "details": status
            }
        except Exception as e:
            results["checks"]["bridge_chat"] = {
                "status": "error",
                "error": str(e)
            }
    else:
        results["checks"]["bridge_chat"] = {
            "status": "not_registered"
        }
    
    # 3. 检查团队管理器
    if _team_manager:
        try:
            teams = _team_manager.list_teams()
            team_statuses = {}
            for team in teams:
                team_statuses[team.team_id] = {
                    "name": team.name,
                    "agent_count": len(team.agents),
                    "status": "ready" if team.agents else "pending"
                }
            results["checks"]["team_manager"] = {
                "status": "ready",
                "teams": team_statuses
            }
        except Exception as e:
            results["checks"]["team_manager"] = {
                "status": "error",
                "error": str(e)
            }
    else:
        results["checks"]["team_manager"] = {
            "status": "not_initialized"
        }
    
    # 4. 计算整体状态
    all_ready = all(
        check.get("status") == "ready"
        for check in results["checks"].values()
    )
    results["overall_status"] = "ready" if all_ready else "pending"
    
    return results
```

@router.get("/validate-pending-start")
async def validate_pending_start():
    """验证所有待启动路径的状态。"""
    results = {
        "timestamp": datetime.utcnow().isoformat(),
        "pending_items": [],
        "validation_results": []
    }
    
    # 1. 检查演进引擎中的待处理项
    if _evolution_engine:
        try:
            pending_items = _evolution_engine.get_evolution_items(status="pending")
            for item in pending_items:
                validation = {
                    "type": "evolution_item",
                    "id": item.get("id"),
                    "title": item.get("title"),
                    "status": item.get("status"),
                    "valid": True,  # 假设有效，实际需要更复杂的验证
                    "message": "待启动项已就绪"
                }
                results["pending_items"].append(validation)
                results["validation_results"].append(validation)
        except Exception as e:
            results["validation_results"].append({
                "type": "evolution_engine",
                "error": str(e),
                "valid": False
            })
    
    # 2. 检查 Build 团队中的待处理任务
    if _build_team:
        try:
            # 假设 build_team 有 pending_tasks 属性
            pending_tasks = getattr(_build_team, 'pending_tasks', [])
            for task in pending_tasks:
                validation = {
                    "type": "build_task",
                    "id": task.get("id"),
                    "description": task.get("description", ""),
                    "status": "pending",
                    "valid": True,
                    "message": "待处理任务已就绪"
                }
                results["pending_items"].append(validation)
                results["validation_results"].append(validation)
        except Exception as e:
            results["validation_results"].append({
                "type": "build_team",
                "error": str(e),
                "valid": False
            })
    
    # 3. 检查执行团队中的待处理反馈
    if _execution_team:
        try:
            pending_feedback = getattr(_execution_team, 'feedback_queue', [])
            for feedback in pending_feedback:
                validation = {
                    "type": "execution_feedback",
                    "id": feedback.get("id"),
                    "title": feedback.get("title", ""),
                    "status": "pending",
                    "valid": True,
                    "message": "待处理反馈已就绪"
                }
                results["pending_items"].append(validation)
                results["validation_results"].append(validation)
        except Exception as e:
            results["validation_results"].append({
                "type": "execution_team",
                "error": str(e),
                "valid": False
            })
    
    # 4. 计算统计
    total = len(results["validation_results"])
    valid = sum(1 for r in results["validation_results"] if r.get("valid"))
    results["statistics"] = {
        "total_pending": total,
        "valid": valid,
        "invalid": total - valid,
        "all_valid": total == valid
    }
    
    return results
```
